def bopen(p):
    h = k32.CreateFileW(p, 0x80000000, 7, None, 3, 0x02000000, None)
    if h in (None, wintypes.HANDLE(-1).value, -1):
        raise OSError(ctypes.get_last_error(), "open failed", p)
    return os.fdopen(msvcrt.open_osfhandle(h, os.O_RDONLY | os.O_BINARY), "rb")
def tail(p, nbytes):
    with bopen(p) as fh:
        fh.seek(0, 2); size = fh.tell(); start = max(0, size - nbytes); fh.seek(start)
        data = fh.read()
    lines = data.split(b"\n")
    if start: lines = lines[1:]
    return size, lines
