def sanitize(cl):
    out = []
    exe = re.match(r'\s*"?([^"]+?\.(?:exe|cmd|py))"?', cl)
    if exe: out.append("exe=" + os.path.basename(exe.group(1)))
    for w in ("ciel_runtime.py cli", "ciel_runtime.py serve", "daemon run", "app-server", "windows_console_guard.py", "codex.js"):
        if w in cl: out.append("has[" + w + "]")
    for f in SAFE:
        if re.search(r'(^|\s)' + re.escape(f) + r'(\s|$)', cl): out.append(f)
    for f in VAL:
        for m in re.finditer(r'(?:^|\s)' + re.escape(f) + r'\s+("([^"]*)"|(\S+))', cl):
            v = m.group(2) if m.group(2) is not None else m.group(3)
            if f in ("--resume", "--plugin-dir", "--bg-pty-host"): v = os.path.basename(v.rstrip("\/"))
            if f == "-c" and "=" in v: v = v.split("=")[0] + "=REDACTED"
            out.append(f + "=" + v[:80])
    out.append("append_system_prompt_present=%s" % ("--append-system-prompt" in cl))
    return " ".join(out)
