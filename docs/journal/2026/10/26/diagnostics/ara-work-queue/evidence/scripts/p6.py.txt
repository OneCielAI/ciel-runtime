NL = chr(10)
def L(ts):
    try: return time.strftime("%m-%dT%H:%M:%S", time.localtime(calendar.timegm(time.strptime(ts[:19], "%Y-%m-%dT%H:%M:%S"))))
    except Exception: return "?"

LT = chr(60)
PREFIXES = ["Stop hook feedback:", LT + "command-name>", LT + "local-command", LT + "channel", LT + "system-reminder>", "[Request interrupted", "Caveat:", "This session is being continued", LT + "task-notification", LT + "bash-"]
STOPRE = re.compile("Stop hook feedback:.*?Ciel Work has ([0-9]+) actionable unfinished item[(]s[)].*?The next item is (n[0-9]+)", re.S)
def classify(s):
    for p in PREFIXES:
        if s.startswith(p):
            if p == "Stop hook feedback:":
                m = STOPRE.match(s)
                return "STOP_FEEDBACK ciel_work actionable=%s next=%s last=%s" % (m.group(1), m.group(2), "last continuation" in s) if m else "STOP_FEEDBACK other len=%d" % len(s)
            return "prefix[%s] len=%d" % (p.replace(LT, "LT"), len(s))
    return "user_prompt len=%d" % len(s)
CHECK = chr(0x2713); WAIT = chr(0x25f7)
MARK = {"[ ]": "pend", "[~]": "prog", "[" + CHECK + "]": "done", "[!]": "blck"}
NODE = re.compile("( *)([[].[]]) (n[0-9]+) ")
QRE = re.compile("(n[0-9]+) [^;]*?[(](pending|in_progress|blocked|done)[)]( [[]held[]])?(?:;|$)")
def text_struct(t):
    out = []
    for line in t.split(NL):
        m = re.match("Work phase: ([a-z_]+)", line)
        if m: out.append("phase=" + m.group(1)); continue
        m = re.match("Active goal progress: ([0-9]+)/([0-9]+)", line)
        if m: out.append("progress=%s/%s" % m.groups()); continue
        m = NODE.match(line)
        if m:
            out.append("%d%s%s%s%s" % (len(m.group(1)) // 2, MARK.get(m.group(2), "?"), m.group(3), "W" if WAIT in line else "", "U" if "[urgent]" in line else "")); continue
        if line.startswith("Queued goals"):
            out.append("queued=" + ",".join("%s:%s%s" % (a, b, "H" if h else "") for a, b, h in QRE.findall(line)))
        if line.startswith("Other goals"):
            out.append("other=" + ",".join("%s:%s" % (a, b) for a, b, h in QRE.findall(line)))
        if line.startswith("Finished goals"):
            out.append("finished=" + ",".join(re.findall("(?:^Finished goals: |; )(n[0-9]+) ", line)))
        m = re.match("(Updated|Added|Removed|Selected|Reordered) (n[0-9]+)", line)
        if m: out.append("op=%s:%s" % m.groups())
    return " ".join(out)[:900]
def json_struct(j):
    c = j.get("continuation") or {}
    q = j.get("queue") or []
    a = j.get("active_root") or {}
    return "JSON en=%s ph=%s appr=%s ac=%s v=%s active=%s:%s%s %s/%s ready=%s cont[act=%s next=%s:%s wait=%s blk=%s held=%s] queue=%s" % (
        j.get("enabled"), j.get("phase"), j.get("approval"), j.get("auto_continue"), j.get("version"), a.get("id"), a.get("status"), "U" if a.get("urgent") else "",
        j.get("done"), j.get("total"), j.get("ready"), c.get("actionable"), (c.get("next") or {}).get("id"), (c.get("next") or {}).get("status"),
        len(c.get("waiting") or []), c.get("blocked"), c.get("held"),
        ",".join("%s:%s%s%s/%s" % (x.get("id"), x.get("status"), "H" if x.get("held") else "", x.get("done"), x.get("total")) for x in q))
